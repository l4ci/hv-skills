def parse_term_entry(body: str) -> dict:
    """Extract definition, aliases, nots from a CONTEXT.md term section body.

    Body is the text after `## <term>` up to the next `## ` heading (the same
    string `iter_topics` yields). Recognized markers in the body:

      - `**Aliases:** a, b, c`  (or `_none_` for empty list)
      - `**Not:** x, y, z`      (optional; absent → empty list)
      - `<!-- YYYY-MM-DD -->`   (date stamp; ignored here, parsed by callers)

    Everything before the first marker line is the definition (whitespace-
    stripped). The HTML date comment is treated as a marker (definition ends
    above it) but its value is not parsed by this helper.

    Returns: {"definition": str, "aliases": list[str], "nots": list[str]}
    """
    aliases: list[str] = []
    nots: list[str] = []
    def_lines: list[str] = []
    seen_marker = False
    for line in body.splitlines():
        stripped = line.strip()
        if stripped.startswith("**Aliases:**"):
            value = stripped[len("**Aliases:**"):].strip()
            if value and value != "_none_":
                aliases = [a.strip() for a in value.split(",") if a.strip()]
            seen_marker = True
            continue
        if stripped.startswith("**Not:**"):
            value = stripped[len("**Not:**"):].strip()
            if value:
                nots = [n.strip() for n in value.split(",") if n.strip()]
            seen_marker = True
            continue
        if stripped.startswith("<!--") and stripped.endswith("-->"):
            seen_marker = True
            continue
        if not seen_marker:
            def_lines.append(line)
    definition = "\n".join(def_lines).strip()
    return {"definition": definition, "aliases": aliases, "nots": nots}


def first_sentence(text: str, max_chars: int = 160) -> str:
    """Return the leading sentence of `text` (up to the first `.`, `!`, or `?`
    followed by whitespace or EOL). If the result exceeds `max_chars`, hard-cut
    at the last word boundary <= max_chars and append `…`. Never raises;
    returns `""` for empty input.

    Used by the CLAUDE.md `## Project Context` block renderer to derive a
    one-line gloss from a term's full definition paragraph.
    """
    if not text:
        return ""
    text = text.strip()
    m = re.search(r"[.!?](?:\s|$)", text)
    sentence = text[:m.end()].rstrip() if m else text
    if len(sentence) <= max_chars:
        return sentence
    cut = sentence[:max_chars].rsplit(" ", 1)[0]
    return cut.rstrip(".,;:") + "…"
```

Create `bin/hv-context-query`:

```bash
#!/usr/bin/env bash
# Print the matching ## <term> section(s) from .hv/CONTEXT.md (single-repo)
# or umbrella + active sub-repo CONTEXT.md (umbrella mode).
# Usage: hv-context-query <term>...
set -euo pipefail
HERE="$(cd "$(dirname "$0")" && pwd)"
. "$HERE/hv-self-locate.sh"
hv_self_locate

# Single-repo path: just delegate to hv-section-query.
if [ ! -d ".hv/contexts" ]; then
  exec "$HERE/hv-section-query" context "$@"
fi

# Umbrella path: union of .hv/CONTEXT.md + (active sub-repo's CONTEXT.md if any).
# Sub-repo entries override umbrella-shared entries with the same name.
SUBREPO=""
SUBREPO=$("$HERE/hv-resolve-repo" 2>/dev/null || true)
PYTHONPATH="$HERE${PYTHONPATH:+:$PYTHONPATH}" python3 - "$SUBREPO" "$@" <<'PY'
import sys
from pathlib import Path
from hvlib import iter_topics

subrepo = sys.argv[1]
wanted = {t.strip().lower() for t in sys.argv[2:]}
if not wanted:
    sys.exit(0)

sources: list[tuple[Path, str]] = []
shared = Path(".hv/CONTEXT.md")
if shared.exists():
    sources.append((shared, "umbrella"))
if subrepo:
    sub = Path(f".hv/contexts/{subrepo}/CONTEXT.md")
    if sub.exists():
        sources.append((sub, subrepo))

# Collect with later sources winning (sub-repo overrides umbrella).
sections: dict[str, tuple[str, str, str]] = {}  # lower_name → (display_name, body, source)
for path, label in sources:
    for name, body in iter_topics(path.read_text()):
        if name.lower() in wanted:
            sections[name.lower()] = (name, body, str(path))

first = True
for key in sorted(sections.keys()):
    name, body, src = sections[key]
    if not first:
        print()
    print(f"> from: {src}")
    print(f"## {name}")
    print(body.rstrip())
    first = False
PY
```

Create `bin/hv-context-index`:

```bash
#!/usr/bin/env bash
# Regenerate the managed <!-- hv-context-start -->...<!-- hv-context-end -->
# block in CLAUDE.md from .hv/CONTEXT.md (and umbrella .hv/contexts/<repo>/
# CONTEXT.md when applicable). Each line: term + (aka aliases) + first-sentence
# gloss. Source files are always listed; in umbrella mode the active sub-repo
# (resolved via hv-resolve-repo) is appended under a ### sub-heading.
# Usage: hv-context-index
set -euo pipefail
HERE="$(cd "$(dirname "$0")" && pwd)"
. "$HERE/hv-self-locate.sh"
hv_self_locate

SUBREPO=""
SUBREPO=$("$HERE/hv-resolve-repo" 2>/dev/null || true)

PYTHONPATH="$HERE${PYTHONPATH:+:$PYTHONPATH}" python3 - "$SUBREPO" <<'PY' | "$HERE/hv-managed-block" context --body-stdin
import sys
from pathlib import Path
from hvlib import iter_topics, parse_term_entry, first_sentence

subrepo = sys.argv[1]

def render_entries(path: Path) -> list[str]:
    if not path.exists():
        return []
    out: list[str] = []
    for name, body in iter_topics(path.read_text()):
        parsed = parse_term_entry(body)
        gloss = first_sentence(parsed["definition"])
        aliases = parsed["aliases"]
        aka = f" *(aka {', '.join(aliases)})*" if aliases else ""
        if gloss:
            out.append(f"- **{name}**{aka} — {gloss}")
        else:
            out.append(f"- **{name}**{aka}")
    return out

heading = "## Project Context"
intro = (
    "Domain terminology lives in `.hv/CONTEXT.md`. Use these canonical "
    "names; if a term you're using conflicts (synonym or drift), call it "
    "out."
)

shared_path = Path(".hv/CONTEXT.md")
shared_entries = render_entries(shared_path)

is_umbrella = Path(".hv/contexts").is_dir()
sub_entries: list[str] = []
if is_umbrella and subrepo:
    sub_path = Path(f".hv/contexts/{subrepo}/CONTEXT.md")
    sub_entries = render_entries(sub_path)

lines = [heading, "", intro, ""]
if not is_umbrella:
    if shared_entries:
        lines.extend(shared_entries)
    else:
        lines.append("- _(no terms yet — run `/hv-context` to capture domain terminology)_")
else:
    if shared_entries:
        lines.extend(shared_entries)
    else:
        lines.append("- _(no umbrella-shared terms yet)_")
    if subrepo:
        lines.append("")
        lines.append(f"### {subrepo}")
        lines.append("")
        if sub_entries:
            lines.extend(sub_entries)
        else:
            lines.append(f"- _(no terms yet for {subrepo} — run `/hv-context` from this sub-repo)_")

sys.stdout.write("\n".join(lines) + "\n")
PY
```

```bash
#!/usr/bin/env bash
# Add or update a term in the appropriate CONTEXT.md.
#
# Usage:
#   hv-context-add <term> --def <text> [--alias <a>,...] [--not <n>,...]
#                  [--repo <name|umbrella>] [--touch]
#
# Single-repo: writes to .hv/CONTEXT.md.
# Umbrella: see Task 7 — flag --repo selects target file; otherwise resolved
# via hv-resolve-repo (falls back to umbrella-shared file when not inside a
# registered sub-repo).
#
# Definition rules:
#   - new term  → insert section alphabetically; today's date stamp
#   - existing  → replace definition; union aliases; union nots; date stamp
#                 preserved unless --touch; --not "" clears the Not list
#
# Refuses (non-zero) when an alias collides with an alias already attached to
# a different term in any in-scope CONTEXT.md file.
set -euo pipefail

TERM=""
DEF=""
ALIASES_RAW=""
NOTS_RAW=""
TARGET_REPO=""
TOUCH="false"
NOTS_PROVIDED="false"

while [ $# -gt 0 ]; do
  case "$1" in
    --def)   DEF="$2"; shift 2 ;;
    --alias) ALIASES_RAW="$2"; shift 2 ;;
    --not)   NOTS_RAW="$2"; NOTS_PROVIDED="true"; shift 2 ;;
    --repo)  TARGET_REPO="$2"; shift 2 ;;
    --touch) TOUCH="true"; shift ;;
    --) shift; break ;;
    -*) echo "error: unknown flag $1" >&2; exit 2 ;;
    *) [ -z "$TERM" ] && TERM="$1" || { echo "error: unexpected positional $1" >&2; exit 2; }; shift ;;
  esac
done

[ -n "$TERM" ] || { echo "usage: hv-context-add <term> --def <text> [--alias …] [--not …] [--repo …] [--touch]" >&2; exit 2; }
[ -n "$DEF" ]  || { echo "error: --def is required" >&2; exit 2; }

HERE="$(cd "$(dirname "$0")" && pwd)"
. "$HERE/hv-self-locate.sh"
hv_self_locate

# Resolve target file. Umbrella behavior is added in Task 7; for now we
# always write to .hv/CONTEXT.md and ignore --repo.
TARGET=".hv/CONTEXT.md"
mkdir -p "$(dirname "$TARGET")"
[ -f "$TARGET" ] || printf '# Context\n\n' > "$TARGET"

PYTHONPATH="$HERE${PYTHONPATH:+:$PYTHONPATH}" python3 - "$TERM" "$DEF" "$ALIASES_RAW" "$NOTS_RAW" "$NOTS_PROVIDED" "$TOUCH" "$TARGET" <<'PY'
import datetime as dt
import re
import sys
from pathlib import Path
from hvlib import iter_topics, parse_term_entry, write_text_atomic

term, definition, aliases_raw, nots_raw, nots_provided, touch_flag, target = sys.argv[1:]
target_path = Path(target)
nots_provided = (nots_provided == "true")
touch = (touch_flag == "true")

def split_csv(s: str) -> list[str]:
    return [x.strip() for x in s.split(",") if x.strip()]

new_aliases = split_csv(aliases_raw)
new_nots = split_csv(nots_raw)

text = target_path.read_text()
existing: dict[str, dict] = {}  # name → {body, parsed, date}
order: list[str] = []
for name, body in iter_topics(text):
    parsed = parse_term_entry(body)
    date_m = re.search(r"<!--\s*(\d{4}-\d{2}-\d{2})\s*-->", body)
    existing[name.lower()] = {
        "name": name,
        "parsed": parsed,
        "date": date_m.group(1) if date_m else None,
    }
    order.append(name)

today = dt.date.today().isoformat()
key = term.lower()

# Alias-conflict check: any of new_aliases also listed as alias of a different term?
for other_key, info in existing.items():
    if other_key == key:
        continue
    other_aliases = {a.lower() for a in info["parsed"]["aliases"]}
    collision = [a for a in new_aliases if a.lower() in other_aliases]
    if collision:
        print(
            f"error: alias '{collision[0]}' is already an alias of term "
            f"'{info['name']}' — refuse to write",
            file=sys.stderr,
        )
        sys.exit(3)

if key in existing:
    prev = existing[key]
    merged_aliases: list[str] = []
    seen: set[str] = set()
    for a in prev["parsed"]["aliases"] + new_aliases:
        al = a.lower()
        if al not in seen:
            seen.add(al)
            merged_aliases.append(a)
    if nots_provided:
        merged_nots: list[str] = []
        seen_n: set[str] = set()
        for n in prev["parsed"]["nots"] + new_nots:
            nl = n.lower()
            if nl not in seen_n:
                seen_n.add(nl)
                merged_nots.append(n)
    else:
        merged_nots = list(prev["parsed"]["nots"])
    date = today if touch else (prev["date"] or today)
    new_def = definition
    new_name = prev["name"]
else:
    merged_aliases = list(new_aliases)
    merged_nots = list(new_nots)
    date = today
    new_def = definition
    new_name = term

# Build the section block.
def build_section(name: str, definition: str, aliases: list[str], nots: list[str], date: str) -> str:
    aliases_line = f"**Aliases:** {', '.join(aliases)}" if aliases else "**Aliases:** _none_"
    parts = [f"## {name}", "", definition, "", aliases_line]
    if nots:
        parts.append(f"**Not:** {', '.join(nots)}")
    parts.extend(["", f"<!-- {date} -->", ""])
    return "\n".join(parts)

new_section = build_section(new_name, new_def, merged_aliases, merged_nots, date)

# Splice into the file. We replace the whole `# Context` body by re-rendering
# in alphabetical order (case-insensitive), preserving each entry's parsed
# fields, then writing the result. Header (`# Context\n\nIntro paragraph…`)
# above the first `## ` is preserved as-is.
m = re.search(r"^## ", text, re.MULTILINE)
header = text[:m.start()] if m else text
if not header.endswith("\n"):
    header += "\n"
if not header.endswith("\n\n"):
    header += "\n"

merged: dict[str, dict] = {**existing}
merged[key] = {
    "name": new_name,
    "parsed": {"definition": new_def, "aliases": merged_aliases, "nots": merged_nots},
    "date": date,
}

ordered_keys = sorted(merged.keys())
sections = [
    build_section(
        merged[k]["name"],
        merged[k]["parsed"]["definition"],
        merged[k]["parsed"]["aliases"],
        merged[k]["parsed"]["nots"],
        merged[k]["date"] or today,
    )
    for k in ordered_keys
]
new_text = header + "\n".join(sections).rstrip() + "\n"
write_text_atomic(target_path, new_text)
PY

# After write, regenerate CLAUDE.md block. (hv-context-index uses cwd to
# resolve any active sub-repo.)
"$HERE/hv-context-index" >/dev/null
```

# hv-context — Capture Project Terminology

Add or refine an entry in `.hv/CONTEXT.md` so future work uses the same canonical name and definition. Terminology drift is silent — sharpening a term once saves every future cycle from re-deciding what it means.

## Step 1 — Preflight

```bash
.hv/bin/hv-preflight
```

See `docs/reference/preflight.md` for exit-code handling.

**Initialize task list.** When `TaskCreate` is loaded (load via `ToolSearch select:TaskCreate,TaskUpdate` if not), create one task per phase below — e.g. `TaskCreate(subject="Resolve term + definition", description="Surface the term name and a one-paragraph definition")`. Mark each `in_progress` when starting and `completed` when its observable outcome lands.

Phases:

1. *Resolve* — term name + definition + optional aliases/nots resolved (Step 2)
2. *Check for conflicts* — existing entry surveyed; user picks keep/replace/merge if found (Step 3)
3. *Write* — `bin/hv-context-add` writes the entry (Step 4)
4. *Index* — `hv-context-add` regenerates the CLAUDE.md block (and CONTEXT-MAP.md in umbrella mode) (Step 4 cont.)
5. *Commit* — single atomic commit (Step 5)

## Step 2 — Resolve Term and Definition

The argument forms determine whether `Step 2` asks any questions:

- **One-shot, fully-specified:** `/hv-context backlog "the canonical project queue (.hv/TODO.md)" --alias "task list,todo list" [--not "session state"] [--repo umbrella|<name>]` — skip to Step 3 with the values from args.
- **Term + definition only:** `/hv-context <term> "<definition>"` — proceed to Step 3 with no aliases/nots; skip the optional-fields prompt.
- **Term only:** `/hv-context <term>` — ask for definition via `AskUserQuestion`; skip aliases/nots unless the user volunteers them.
- **No args:** ask for term and definition in two single-question turns. **Don't ask what the codebase already implies** — skip aliases/nots prompts unless context (`KNOWLEDGE.md` topics, recent commits, current `## Project Context` block in `CLAUDE.md`) makes a candidate alias obvious; even then, propose inline rather than blocking on a question.

In umbrella mode, target file resolution mirrors `hv-context-add` defaults:

- `--repo umbrella` → `.hv/CONTEXT.md`
- `--repo <name>` → `.hv/contexts/<name>/CONTEXT.md`
- omitted, cwd inside a registered sub-repo → that sub-repo's file (resolved via `hv-resolve-repo`)
- omitted, cwd at umbrella root → ask once: *"Capture this as an umbrella-shared term, or scope to a sub-repo?"* (`AskUserQuestion`, options: *Umbrella-shared (Recommended)*, one option per registered sub-repo)

## Step 3 — Check for Conflicts

Run `.hv/bin/hv-context-query <term>` to see whether the term already exists. If it does, surface the existing entry and ask the user via `AskUserQuestion`:

- **Header:** `"Term"`
- **Question:** *"Term `<term>` already exists. What should we do?"*
- **Options** (single-select):
  1. *"Replace definition, union aliases (Recommended)"* — *"Replace the definition with your new wording, merge new aliases into the existing list, preserve the original date stamp."*
  2. *"Replace definition + bump date (`--touch`)"* — *"Same as above but mark the entry as freshly refined."*
  3. *"Cancel"* — *"Skip — nothing is written."*

Show the existing entry inline above the question.

If the term doesn't exist, also check for **alias collisions** by scanning the `## Project Context` block in `CLAUDE.md` for any of the new aliases. If one matches another term's alias, surface the collision and stop — `hv-context-add` will refuse with exit 4 anyway, but a pre-check is friendlier:

> *"Alias `<x>` is already an alias of term `<other-term>`. Either pick a different alias, or refine the existing term instead — what would you like to do?"*

(`AskUserQuestion`: *Use a different alias*, *Refine `<other-term>` instead*, *Cancel*.)

## Step 4 — Write and Index

Invoke `bin/hv-context-add` with the resolved arguments. Example calls:

**Single-repo, new term:**
```bash
.hv/bin/hv-context-add backlog \
  --def "The canonical project queue (.hv/TODO.md). Items are zero-padded IDs ([B01]/[F01]/[T01])." \
  --alias "task list,todo list"
```

**Umbrella, sub-repo:**
```bash
.hv/bin/hv-context-add agent-loop \
  --def "The repeating prompt-stage loop in repo-a." \
  --repo repo-a
```

**Update existing, bump date:**
```bash
.hv/bin/hv-context-add backlog \
  --def "<refined definition>" \
  --alias "queue" \
  --touch
```

`hv-context-add` regenerates the `## Project Context` managed block (and `.hv/CONTEXT-MAP.md` in umbrella mode) before exiting.

## Step 5 — Commit

Single atomic commit:

```bash
git add .hv/CONTEXT.md .hv/CONTEXT-MAP.md .hv/contexts/ CLAUDE.md
git commit -m "docs(context): add <term>"
```

For an update use `docs(context): refine <term>`. The user is asked once at the end via the standard hv-skills "commit message OK?" flow (matches `/hv-decide`). Don't include attribution beyond what hv-skills already adds.

## Step 6 — Confirm

Tell the user, in one compact block, what was captured:

```
Captured "<term>" into .hv/CONTEXT.md.
  aliases: <a, b>     (or "none" if empty)
  nots:    <x, y>     (or omit the line if empty)

Updated CLAUDE.md ## Project Context block.
```

In umbrella mode, prepend the target file: *"Captured into .hv/contexts/repo-a/CONTEXT.md."*

## Key Principles

- **Explicit signal only.** Never infer a write from implicit term usage — only when the user clearly defines or refines a term. The signal phrases are listed in the spec; this skill is the deliberate entry point for everything else.
- **Don't ask what the code can answer.** Aliases, nots, target file — derive from args, cwd, and `hv-resolve-repo` before asking. Ask only when ambiguous.
- **One sentence's-worth of definition is fine.** The first sentence becomes the gloss in the CLAUDE.md block; longer definitions stay in CONTEXT.md but the gloss must stand alone.
- **Aliases are user-stated only.** Don't propose aliases the user didn't ask for. Conflict-call-out (during a `/hv-work` cycle) can suggest "you used X — CONTEXT calls this Y" but only `/hv-context` writes them.
- **No verifier.** Manual confirmation is the verification.

# Capturing terminology — `/hv-context` and `CONTEXT.md`

`.hv/CONTEXT.md` is the project's canonical glossary — one entry per domain term, with a definition, optional aliases, and optional "not" clarifications. The AI consults it before answering or brainstorming so terms stay consistent across cycles, and flags inline when your wording conflicts with the canonical name or its meaning.

This is distinct from:

- **`.hv/KNOWLEDGE.md`** (`/hv-learn`) — passive *gotchas, conventions, constraints*. What you wish you'd known.
- **`.hv/DECISIONS.md`** (`/hv-decide`) — active *boundaries with forbids/permits*. What the project has committed to.
- **`.hv/MAP.md`** (`/hv-map`) — *subsystems*. Where things live.
- **`.hv/CONTEXT.md`** — *vocabulary*. What things are called.

## Capturing a term

`/hv-context` is the deliberate entry point. Useful invocation forms:

```text
/hv-context backlog "the canonical project queue (.hv/TODO.md). Items are zero-padded IDs."
/hv-context backlog "<def>" --alias "task list,todo list"
/hv-context backlog "<refined def>" --touch
/hv-context agent-loop "<def>" --repo repo-a       # umbrella sub-repo
/hv-context session "<def>" --repo umbrella        # umbrella shared
```

The skill also writes inline during normal conversation when you produce an unambiguous definitional signal:

- *"let's call this X"* / *"the canonical name for this is X"*
- *"by X I mean Y"* / *"X means Y"* / *"X is our term for Y"*
- *"refine X — it should mean Y"* (existing term update)
- Direct command: *"save X as Y"* / *"add X to context"*

If your wording is ambiguous (you mention a term but don't define it), the skill won't write; it may ask once for clarification. There is no end-of-session sweep — terms not captured at the moment they resolve are captured later via explicit `/hv-context` invocations.

## Conflict call-out

When the AI sees you using a term that conflicts with `CONTEXT.md`, it flags the conflict inline:

- **Synonym** — *"You said 'task list' — this project's CONTEXT calls this **backlog**. Want me to use that going forward, or run `/hv-context` to update the canonical name?"*
- **Drift** — *"You're using **decision** to mean a soft preference. CONTEXT defines **decision** as a hard boundary captured in DECISIONS.md. Did you mean a learning, or run `/hv-context` to refine the term?"*

These are one-line interruptions, not blocking. You answer in flow.

## Aliases and "not" clarifications

Aliases are user-stated synonyms — informal phrasings that map to the canonical term. They drive synonym detection. Add them when:

- The term has a common informal name you want flagged ("task list" → **backlog**).
- A previous name was deprecated and you want to map old usages to the new term.

Don't fabricate aliases. If you didn't ever call it that, don't list it.

The optional `**Not:**` line distinguishes the term from a near-miss concept that is *not* the term. Use it sparingly — only when the contrast is load-bearing.

## Umbrella mode

In umbrella projects, glossaries split:

- `.hv/CONTEXT.md` — terms shared across all sub-repos (cross-cutting domain concepts).
- `.hv/contexts/<repo>/CONTEXT.md` — terms scoped to one sub-repo.
- `.hv/CONTEXT-MAP.md` — auto-generated pointer index.

`/hv-context` resolves the target file from your cwd: inside a sub-repo it writes there; at the umbrella root it asks once whether the term is umbrella-shared. Override with `--repo umbrella` or `--repo <name>`. See `docs/usage/umbrella-mode.md` for more.

# `.hv/CONTEXT.md` file format

Project-level domain glossary. One entry per term. Source-of-truth for the always-on `## Project Context` block in `CLAUDE.md` (regenerated by `bin/hv-context-index`).

## Layout

```markdown
# Context

<intro paragraph from hv-bootstrap>

## <term>

<one-paragraph canonical definition — single paragraph, complete sentences>

**Aliases:** <comma-separated synonyms, or `_none_`>
**Not:** <comma-separated near-misses>

<!-- 2026-05-10 -->
```

## Field rules

- **Heading.** `## <term>` — lowercase by convention; case-sensitive in source but case-insensitive for query/conflict matching.
- **Definition.** Free paragraph. Plain prose; no bullets, no nested headings. The first sentence becomes the gloss in the CLAUDE.md block, so write definitions whose first sentence stands alone.
- **Aliases.** Comma-separated. Use `_none_` when there are no aliases (the line itself is always present — it's the regex anchor for synonym detection).
- **Not.** Optional. Omit the line entirely when there's nothing to distinguish.
- **Date stamp.** HTML comment `<!-- YYYY-MM-DD -->`. Set on creation; preserved on update unless `hv-context-add --touch` is passed.

## Ordering

Terms appear alphabetically (case-insensitive) within the file. `bin/hv-context-add` re-renders the body in alphabetical order on every write to keep ordering canonical.

The header section (`# Context` + intro paragraph above the first `## ` heading) is preserved verbatim across writes.

## Conflict semantics

`hv-context-add` refuses with exit code 4 when:

- A term name collides with another term whose name is the same case-insensitively (handled as an update, not a refusal).
- A new alias collides with an alias already attached to a *different* term in the same scope (single-repo: just `.hv/CONTEXT.md`; umbrella: target file + umbrella-shared file).

## Umbrella

In umbrella mode the file lives at:

| Scope | Path |
|---|---|
| Umbrella-shared (cross-repo terms) | `.hv/CONTEXT.md` |
| Per-sub-repo | `.hv/contexts/<repo>/CONTEXT.md` |
| Auto-generated pointer index | `.hv/CONTEXT-MAP.md` |

`bin/hv-context-map` regenerates `CONTEXT-MAP.md` after every umbrella-mode write.

## Helpers

| Helper | Purpose |
|---|---|
| `bin/hv-context-query <term>...` | Print matching `## <term>` sections (umbrella-aware) |
| `bin/hv-context-index` | Regenerate `## Project Context` block in `CLAUDE.md` |
| `bin/hv-context-add <term> --def <text> [--alias …] [--not …] [--repo …] [--touch]` | Insert or update a term |
| `bin/hv-context-map` | Regenerate `.hv/CONTEXT-MAP.md` (umbrella only) |

See `docs/reference/cli-helpers.md` for full helper signatures.
```
